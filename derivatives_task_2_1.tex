\maketitle

The ice thickness is
\[
q(H,T)=\sqrt{H^2+C\,T\,\Delta t},
\]
where
\[
C=1.44\times 10^{-8}\ \si{\meter\squared\per\kelvin\per\second},
\qquad
\Delta t=5(24)(3600)=432000\ \si{\second}.
\]

Define
\[
A=C\Delta t=0.0062208.
\]
Then
\[
q(H,T)=\sqrt{H^2+AT}.
\]

At the mean values
\[
H=\mu_{H_0}=0.20\ \si{\meter},
\qquad
T=\mu_{\Delta T}=10\ \si{\kelvin},
\]
the function evaluates to
\[
q=\sqrt{(0.20)^2+(0.0062208)(10)}
  =0.319699859\ \si{\meter}.
\]

\section*{Derivatives}

\[
\frac{\partial q}{\partial H}
=\frac{H}{\sqrt{H^2+AT}}
=\frac{H}{q}
\]

\[
\frac{\partial^2 q}{\partial H^2}
=\frac{AT}{\left(H^2+AT\right)^{3/2}}
=\frac{AT}{q^3}
\]

\[
\frac{\partial q}{\partial T}
=\frac{A}{2\sqrt{H^2+AT}}
=\frac{A}{2q}
\]

\[
\frac{\partial^2 q}{\partial T^2}
=-\frac{A^2}{4\left(H^2+AT\right)^{3/2}}
=-\frac{A^2}{4q^3}
\]

\[
\frac{\partial^2 q}{\partial H\,\partial T}
=-\frac{AH}{2\left(H^2+AT\right)^{3/2}}
=-\frac{AH}{2q^3}.
\]

Because \(q\) is smooth,
\[
\frac{\partial^2 q}{\partial H\,\partial T}
=\frac{\partial^2 q}{\partial T\,\partial H}.
\]

\section*{Numerical values at the mean}

\[
\boxed{
\begin{aligned}
\left.\frac{\partial q}{\partial H}\right|_0
  &=0.625586763,\\[2mm]
\left.\frac{\partial^2 q}{\partial H^2}\right|_0
  &=1.903789397,\\[2mm]
\left.\frac{\partial q}{\partial T}\right|_0
  &=0.009729125,\\[2mm]
\left.\frac{\partial^2 q}{\partial T^2}\right|_0
  &=-2.960773\times 10^{-4},\\[2mm]
\left.\frac{\partial^2 q}{\partial H\,\partial T}\right|_0
  &=-0.019037894.
\end{aligned}
}
\]
